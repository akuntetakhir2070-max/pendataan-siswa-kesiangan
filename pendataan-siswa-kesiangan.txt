    if(records.length === 0) initDummyData();

  function showPage(pageId) {
    document.querySelectorAll('.page').forEach(el => el.classList.remove('active'));
    document.getElementById('page-' + pageId).classList.add('active');

    document.querySelectorAll('.nav-item').forEach(el => el.classList.remove('active'));
    document.getElementById('nav-' + pageId).classList.add('active');

  }

  function populateClassSelects() {
    const select1 = document.getElementById('input-kelas');
    const select2 = document.getElementById('filter-kelas');
    let opts = '';
    MOCK_CLASSES.forEach(c => opts += `<option value="${c}">${c}</option>`);
    select1.innerHTML = '<option value="">Pilih Kelas</option>' + opts;
    select2.innerHTML = '<option value="">Semua Kelas</option>' + opts;
  }

  function updateStudentList() {
    const kelas = document.getElementById('input-kelas').value;
    const nameSelect = document.getElementById('input-nama');
    
    if (!kelas || !STUDENT_DATA[kelas]) {
      nameSelect.innerHTML = '<option value="">Pilih Kelas Terlebih Dahulu</option>';
      nameSelect.disabled = true;
      return;
    }

    nameSelect.disabled = false;
    let opts = '<option value="">Pilih Nama Siswa</option>';
    STUDENT_DATA[kelas].forEach(name => {
      opts += `<option value="${name}">${name}</option>`;
    });
    nameSelect.innerHTML = opts;
  }

  function startClock() {
    const tick = () => {
      const now = new Date();
      const h = String(now.getHours()).padStart(2, '0');
      const m = String(now.getMinutes()).padStart(2, '0');
      const s = String(now.getSeconds()).padStart(2, '0');
      const timeStr = `${h}:${m}:${s}`;
      document.getElementById('clock').textContent = timeStr;
      document.getElementById('form-clock').textContent = timeStr;
    };
    tick();
    clockInterval = setInterval(tick, 1000);
  }

  function updateDateDisplay() {
    const now = new Date();
    const options = { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' };
    const dateStr = now.toLocaleDateString('id-ID', options);
    document.getElementById('topbar-date').textContent = dateStr;
    document.getElementById('form-date').textContent = dateStr;
  }

  function calculateDuration(time) {
    const [h1, m1] = limitTime.split(':').map(Number);
    const [h2, m2] = time.split(':').map(Number);
    let diff = (h2 * 60 + m2) - (h1 * 60 + m1);
    return diff > 0 ? diff : 0;
  }

  function refreshData() {
    renderDashboard();
    if (document.getElementById('page-record').classList.contains('active')) {
      renderRecords();
    }
  }

  function renderDashboard() {
    const todayStr = new Date().toLocaleDateString('en-CA');
    const weekAgo = new Date();
    weekAgo.setDate(weekAgo.getDate() - 7);

    let countToday = 0;
    let countWeek = 0;
    let classesTainted = new Set();
    let classCounts = {};
    let studentCounts = {};

    records.forEach(r => {
      if(r.date === todayStr) countToday++;
      if(new Date(r.date) >= weekAgo) countWeek++;
      classesTainted.add(r.cls);

      classCounts[r.cls] = (classCounts[r.cls] || 0) + 1;
      studentCounts[r.name] = studentCounts[r.name] || { name: r.name, cls: r.cls, count: 0 };
      studentCounts[r.name].count++;
    });

    document.getElementById('stat-total').textContent = records.length;
    document.getElementById('stat-today').textContent = countToday;
    document.getElementById('stat-week').textContent = countWeek;
    document.getElementById('stat-classes').textContent = classesTainted.size;

    // Render Bar Chart
    const barContainer = document.getElementById('bar-chart');
    const sortedClasses = Object.keys(classCounts).sort((a,b) => classCounts[b] - classCounts[a]).slice(0, 5);
    const maxCount = sortedClasses.length ? classCounts[sortedClasses[0]] : 1;
    
    let barHTML = '';
    sortedClasses.forEach(cls => {
      const cnt = classCounts[cls];
      const heightPercent = Math.max((cnt / maxCount) * 100, 5); // min 5% height
      barHTML += `
        <div class="bar-item">
          <div class="bar-count">${cnt}</div>
          <div class="bar" style="height:${heightPercent}%; background:var(--primary);"></div>
          <div class="bar-label" title="${cls}">${cls.split(' ')[1] || cls}</div>
        </div>
      `;
    });
    if(sortedClasses.length === 0) barHTML = '<div style="color:var(--muted);font-size:12px;">Data statistik kosong</div>';
    barContainer.innerHTML = barHTML;
    document.getElementById('chart-total').textContent = records.length;

    // Render Top Students
    const topStudents = Object.values(studentCounts).sort((a,b) => b.count - a.count).slice(0, 4);
    let topHTML = '';
    topStudents.forEach((st, idx) => {
      const rclass = idx === 0 ? 'rank-1' : (idx === 1 ? 'rank-2' : (idx === 2 ? 'rank-3' : 'rank-n'));
      topHTML += `
        <div class="top-student-row">
          <div style="display:flex;align-items:center;gap:12px;">
            <div class="rank-badge ${rclass}">${idx+1}</div>
            <div class="student-info">
              <div class="name">${st.name}</div>
              <div class="kelas">${st.cls}</div>
            </div>
          </div>
          <div class="count-badge">${st.count}x</div>
        </div>
      `;
    });
    if(topStudents.length === 0) topHTML = '<div style="padding:10px;text-align:center;color:var(--muted);font-size:12px;">Belum ada data rekapan</div>';
    document.getElementById('top-students-list').innerHTML = topHTML;

    // Render Recent History Table
    const recent = [...records].sort((a,b) => b.timestamp - a.timestamp).slice(0, 5);
    const recentTbody = document.getElementById('recent-tbody');
    let recHTML = '';
    recent.forEach(r => {
      const dur = calculateDuration(r.time);
      const studentTotal = studentCounts[r.name] ? studentCounts[r.name].count : 1;
      
      recHTML += `
        <tr>
          <td class="td-name" style="padding-left:20px;">${r.name}</td>
          <td><span class="badge badge-secondary">${r.cls}</span></td>
          <td class="td-time">${r.time}</td>
          <td><span style="color:var(--destructive);font-weight:700;">+${dur}m</span></td>
          <td>${studentTotal}x</td>
          <td style="color:var(--muted);font-size:12px;">${r.date}</td>
        </tr>
      `;
    });
    if(recent.length === 0) recHTML = '<tr><td colspan="6" style="text-align:center;padding:40px;color:var(--muted);">Belum ada riwayat terbaru</td></tr>';
    recentTbody.innerHTML = recHTML;
    document.getElementById('total-badge').textContent = `${records.length} total`;
  }

  function renderRecords() {
    const fName = document.getElementById('filter-nama').value.toLowerCase();
    const fClass = document.getElementById('filter-kelas').value;
    const fDate = document.getElementById('filter-date').value;

    let filtered = records.filter(r => {
      let match = true;
      if(fName && !r.name.toLowerCase().includes(fName)) match = false;
      if(fClass && r.cls !== fClass) match = false;
      if(fDate && r.date !== fDate) match = false;
      return match;
    });

    filtered.sort((a,b) => b.timestamp - a.timestamp);

    const tbody = document.getElementById('records-tbody');
    let html = '';
    filtered.forEach(r => {
      html += `
        <tr>
          <td class="td-name" style="padding-left:16px;">${r.name}</td>
          <td><span class="badge badge-secondary">${r.cls}</span></td>
          <td class="td-time">${r.time}</td>
          <td style="color:var(--muted);font-size:12px;">${r.date}</td>
          <td class="td-muted" title="${r.reason}">${r.reason || '-'}</td>
          <td style="text-align:right;padding-right:16px;">
            <button class="btn-del" onclick="deleteRecord(${r.id})" title="Hapus Data">
              <svg fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M19 7l-.867 12.142A2 2 0 0116.138 21H7.862a2 2 0 01-1.995-1.858L5 7m5 4v6m4-6v6m1-10V4a1 1 0 00-1-1h-4a1 1 0 00-1 1v3M4 7h16"/></svg>
            </button>
          </td>
        </tr>
      `;
    });
    if(filtered.length === 0) html = '<tr><td colspan="6" style="text-align:center;padding:40px;color:var(--muted);">Tidak ada data sesuai filter</td></tr>';
    tbody.innerHTML = html;
  }

  function addRecord() {
    const nama = document.getElementById('input-nama').value.trim();
    const kelas = document.getElementById('input-kelas').value;
    const alasan = document.getElementById('input-alasan').value.trim();

    if(!nama || !kelas) {
      showBanner('Nama Siswa dan Kelas wajib diisi!', true);
      return;
    }

    const now = new Date();
    const h = String(now.getHours()).padStart(2, '0');
    const m = String(now.getMinutes()).padStart(2, '0');
    const timeStr = `${h}:${m}`;
    const dateStr = now.toLocaleDateString('en-CA');

    const newRecord = {
      id: Date.now(),
      name: nama,
      cls: kelas,
      time: timeStr,
      date: dateStr,
      timestamp: now.getTime(),
      reason: alasan
    };

    records.push(newRecord);
    refreshData();

    // Reset form
    document.getElementById('input-kelas').value = '';
    updateStudentList();
    document.getElementById('input-alasan').value = '';

    showBanner(`${nama} berhasil didata!`, false);
  }

  function openModal(id) {
    document.getElementById(id).classList.add('open');
  }
  
  function closeModal(id) {
    document.getElementById(id).classList.remove('open');
    if(id === 'modal-delete-record') recordToDeleteId = null;

  function deleteRecord(id) {
    recordToDeleteId = id;
    openModal('modal-delete-record');
  }

  function confirmDeleteRecord() {
    if(recordToDeleteId !== null) {
      records = records.filter(r => r.id !== recordToDeleteId);
      refreshData();
      closeModal('modal-delete-record');
    }
  }
